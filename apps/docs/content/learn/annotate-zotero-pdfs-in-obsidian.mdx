import { Check, Fix, Step } from "@/components/learn/lesson-mdx";
import { UiLabel } from "@/components/ui-label";
import { m } from "@/paraglide/messages.js";

<Step n={1}>

In <UiLabel name={m.library_view_name()} />, select Sandve et al. 2013, then <UiLabel name={m.library_view_menu_open_pdf()} /> on its card.

<Check>
  The PDF opens with dimmed annotation tools, and the sidebar reads{" "}
  <UiLabel name={m.annot_view_header_following_active_tab()} /> ·{" "}
  <UiLabel name={m.annot_view_header_reading_only()} />.
</Check>

</Step>

<Step n={2}>

Select the sidebar heading, then <UiLabel name={m.capability_enable_editing()} />.

<Check>
  The PDF toolbar reads <UiLabel name={m.capability_authorizing()} />, and
  Zotero displays a dialog with **Always Allow**.
</Check>

</Step>

<Step n={3}>

In the dialog, select **Always Allow**.

<Check>
  The notice <UiLabel name={m.notice_zotero_editing_enabled()} /> appears, and
  the tools appear in colour.
</Check>

<Fix title="The tools stay dimmed">
  Repeat Step 2 and select **Always Allow**.
</Fix>

</Step>

<Step n={4}>

Drag across a sentence in the PDF.

<Check>
  A toolbar appears under the selection with{" "}
  <UiLabel name={m.pdf_toolbar_highlight()} />,{" "}
  <UiLabel name={m.pdf_toolbar_underline()} />, four colours, and{" "}
  <UiLabel name={m.annot_view_card_add_comment()} />.
</Check>

</Step>

<Step n={5}>

Select <UiLabel name={m.annot_view_card_add_comment()} />, type a comment for Chapter 3, and select <UiLabel name={m.pdf_toolbar_highlight()} />.

<Check>
  The passage turns yellow, and a card showing the page, excerpt, and comment
  appears in the sidebar.
</Check>

</Step>

<Step n={6}>

Switch to Zotero and open the same PDF.

<Check>The PDF in Zotero shows the same yellow highlight and comment.</Check>

</Step>

<Step n={7}>

In Obsidian, open the note for Sandve et al. 2013.

<Check>The **Annotations** section shows a new card for the highlight.</Check>

</Step>
