import { Check, Fix, Step } from "@/components/learn/lesson-mdx";
import { UiLabel } from "@/components/ui-label";
import { m } from "@/paraglide/messages.js";

<Step n={1}>

In the <UiLabel name={m.library_view_name()} /> tab, select Sandve et al. 2013, then select <UiLabel name={m.library_view_menu_open_pdf()} /> on its card.

<Check>
  The PDF opens. The annotation tools at the top right of the PDF are dimmed,
  and the sidebar heading reads{" "}
  <UiLabel name={m.annot_view_header_following_active_tab()} /> ·{" "}
  <UiLabel name={m.annot_view_header_reading_only()} />.
</Check>

</Step>

<Step n={2}>

Select the heading of the sidebar on the right. In the menu, select <UiLabel name={m.capability_enable_editing()} />. You give this permission once on each computer.

<Check>
  The PDF toolbar reads <UiLabel name={m.capability_authorizing()} />. Zotero
  shows a dialog asking whether ZotLit for Obsidian may change the Zotero
  library, with the buttons **Deny**, **Always Allow**, and **Allow**.
</Check>

</Step>

<Step n={3}>

In the Zotero dialog, select **Always Allow**.

<Check>
  In Obsidian, the notice <UiLabel name={m.notice_zotero_editing_enabled()} />{" "}
  appears. The annotation tools are in colour.
</Check>

<Fix title="The tools stay dimmed">
  You chose **Allow**. Do step 2 again and choose **Always Allow**.
</Fix>

</Step>

<Step n={4}>

Drag across a sentence in the PDF.

<Check>
  A small toolbar appears under the selection, with{" "}
  <UiLabel name={m.pdf_toolbar_highlight()} />,{" "}
  <UiLabel name={m.pdf_toolbar_underline()} />, four colours, and{" "}
  <UiLabel name={m.annot_view_card_add_comment()} />.
</Check>

</Step>

<Step n={5}>

Select <UiLabel name={m.annot_view_card_add_comment()} />, type a comment for Chapter 3, then select <UiLabel name={m.pdf_toolbar_highlight()} /> under the comment.

<Check>
  The passage turns yellow. A card for it appears in the sidebar on the right,
  with the page, the passage, and the comment.
</Check>

</Step>

<Step n={6}>

Switch to Zotero and open the same PDF.

<Check>
  The highlight and its comment are there. Each highlight is saved in the Zotero
  library, so Zotero and Obsidian show the same one.
</Check>

</Step>

<Step n={7}>

Switch back to Obsidian and open the literature note for Sandve et al. 2013 from Lesson 1.

<Check>
  The **Annotations** section has a new card for the highlight. You ran no
  update.
</Check>

</Step>
