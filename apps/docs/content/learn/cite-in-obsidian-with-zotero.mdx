import { Check, Fix, Step } from "@/components/learn/lesson-mdx";
import { UiLabel } from "@/components/ui-label";
import { m } from "@/paraglide/messages.js";

<Step n={1}>

In **Settings** > **ZotLit** > <UiLabel name={m.settings_page_citations()} />, select <UiLabel name={m.settings_citation_engine_install()} /> next to <UiLabel name={m.settings_citation_engine_name()} />.

<Check>
  The row reads{" "}
  <UiLabel
    name={m.settings_citation_engine_status_installed({ version: "3.10" })}
  />{" "}
  (the version can be higher).
</Check>

</Step>

<Step n={2}>

In `Chapter 3 - Presenting the results`, place the cursor and type `[@sandve`.

<Check>
  A citation menu opens, showing **Sandve et al. (2013), PLOS Computational
  Biology, 9(10), e1003285**.
</Check>

<Fix title="The paper is not in the list">
  A paper with no citation key cannot be cited. Install Better BibTeX in Zotero
  to give every paper a key.
</Fix>

</Step>

<Step n={3}>

Press Enter to choose the paper.

<Check>The draft displays `[@sandve2013]` at the cursor.</Check>

</Step>

<Step n={4}>

Type `, p. 2` before the closing bracket, then move the cursor away.

<Check>The citation formats as **(Sandve et al. 2013, 2)**.</Check>

<Fix title="It still reads [@sandve2013, p. 2]">
  Install the engine in Step 1.
</Fix>

</Step>

<Step n={5}>

Point at the formatted citation.

<Check>A preview card opens with the full Sandve 2013 reference.</Check>

</Step>

<Step n={6}>

Open the PDF beside your draft. Drag the page label of a highlight, such as **p. 2**, from the sidebar into the draft.

<Check>
  The draft shows a quote block titled **Page 2** with the highlighted text,
  citation, and note.
</Check>

</Step>
