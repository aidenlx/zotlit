import { Check, Fix, Step } from "@/components/learn/lesson-mdx";
import { Command } from "@/components/command";
import { UiLabel } from "@/components/ui-label";
import { m } from "@/paraglide/messages.js";

<Step n={1}>

删除仅供自己查看的内容（例如待办清单），因为笔记中的所有内容都会导出到 Word 中。

<Check>草稿中仅保留你希望出现在 Word 文件中的内容。</Check>

</Step>

<Step n={2}>

按 Ctrl+P（Mac 上按 Cmd+P）并运行 <Command inline name={m.command_pandoc_export_name()} />。

<Check>
  打开 <UiLabel name={m.pandoc_export_title()} /> 对话框，其中{" "}
  <UiLabel name={m.pandoc_export_format_name()} /> 设为{" "}
  <UiLabel name={m.pandoc_export_format_docx()} />。
</Check>

</Step>

<Step n={3}>

在 <UiLabel name={m.pandoc_export_style_name()} /> 中选择 **APA Style 7th edition**。

<Check>
  该设置下方的文字显示为{" "}
  <UiLabel
    name={m.pandoc_export_style_changed({
      style: "Default (Chicago author-date)",
    })}
  />
</Check>

<Fix title="列表中未找到该样式">
  在 Zotero 中安装该样式，然后重新打开对话框。
</Fix>

</Step>

<Step n={4}>

点击 <UiLabel name={m.pandoc_export_confirm()} />。

<Check>
  提示通知中显示了草稿对应的 Word 文件名（此处为{" "}
  <UiLabel
    name={m.notice_pandoc_export_done({
      file: "Chapter 3 - Presenting the results.docx",
      style: "APA Style 7th edition",
    })}
  />
  ）。
</Check>

</Step>

<Step n={5}>

在访达（Mac）或文件资源管理器（Windows）中，打开草稿所在文件夹中的 Word 文件。

<Check>
  Word 文件中显示排版好的引文（例如 **(Sandve et al., 2013, p. 2)**）以及
  **References** 参考文献列表。
</Check>

</Step>
