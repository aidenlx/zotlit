import { Check, Fix, Step } from "@/components/learn/lesson-mdx";
import { Command } from "@/components/command";
import { UiLabel } from "@/components/ui-label";
import { m } from "@/paraglide/messages.js";

<Step n={1}>

Remove lines meant only for you, such as a to-do list, because everything in the note exports to Word.

<Check>The draft contains only what you want in the Word file.</Check>

</Step>

<Step n={2}>

Press Ctrl+P (Cmd+P on Mac) and run <Command inline name={m.command_pandoc_export_name()} />.

<Check>
  The dialog <UiLabel name={m.pandoc_export_title()} /> opens with{" "}
  <UiLabel name={m.pandoc_export_format_name()} /> set to{" "}
  <UiLabel name={m.pandoc_export_format_docx()} />.
</Check>

</Step>

<Step n={3}>

In <UiLabel name={m.pandoc_export_style_name()} />, choose **APA Style 7th edition**.

<Check>
  The line under the setting reads{" "}
  <UiLabel
    name={m.pandoc_export_style_changed({
      style: "Default (Chicago author-date)",
    })}
  />
</Check>

<Fix title="The style is not in the list">
  Install the style in Zotero and reopen the dialog.
</Fix>

</Step>

<Step n={4}>

Select <UiLabel name={m.pandoc_export_confirm()} />.

<Check>
  The notice appears:{" "}
  <UiLabel
    name={m.notice_pandoc_export_done({
      file: "Chapter 3 - Presenting the results.docx",
      style: "APA Style 7th edition",
    })}
  />
</Check>

</Step>

<Step n={5}>

In Finder (Mac) or File Explorer (Windows), open the Word file in the draft’s folder.

<Check>
  The Word file shows formatted citations such as **(Sandve et al., 2013, p.
  2)** and a **References** list.
</Check>

</Step>
