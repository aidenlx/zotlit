import { Check, Fix, Step } from "@/components/learn/lesson-mdx";
import { UiLabel } from "@/components/ui-label";
import { m } from "@/paraglide/messages.js";

<Step n={1}>

In <UiLabel name={m.library_view_name()} />, select <UiLabel name={m.annot_view_save_as_query_file()} /> (page-plus icon, top right of the list). Type `Reading plan` as the name and select <UiLabel name={m.annot_view_save_action()} />.

<Check>The file `Reading plan` opens in a new tab, listing your papers.</Check>

</Step>

<Step n={2}>

Select <UiLabel name={m.query_view_views()} /> at the top left, then the arrow at the end of the row (<UiLabel name={m.query_view_configure_view()} />). Set <UiLabel name={m.query_view_layout()} /> to <UiLabel name={m.query_view_layout_table()} />.

<Check>Your papers appear as rows in a table.</Check>

</Step>

<Step n={3}>

Select <UiLabel name={m.query_view_properties()} />, type `reading-status`, and select <UiLabel name={m.query_view_properties_add_vault({ name: "reading-status" })} />.

<Check>
  A column titled **reading-status (note)** appears, showing{" "}
  <UiLabel name={m.query_view_vault_no_note()} /> for papers without a note.
</Check>

<Fix title="The column header does not say “(note)”">
  Select <UiLabel name={m.query_view_properties()} />, clear that check box, and
  select{" "}
  <UiLabel
    name={m.query_view_properties_add_vault({ name: "reading-status" })}
  />
  .
</Fix>

</Step>

<Step n={4}>

Select a paper’s reading status cell, type `to-read`, and press Enter.

<Check>
  The notice appears: <UiLabel name={m.notice_created_note()} /> The cell shows
  `to-read`.
</Check>

</Step>

<Step n={5}>

Mark another paper as `reading` and a finished paper as `done`.

<Check>
  As you type, the cell suggests statuses you used before. The rows now show
  statuses.
</Check>

</Step>

<Step n={6}>

Select <UiLabel name={m.query_view_filter()} />, set the first box to **reading-status (note)**, the second to **is**, and type `to-read` in the third.

<Check>The table shows only papers marked `to-read`.</Check>

</Step>
