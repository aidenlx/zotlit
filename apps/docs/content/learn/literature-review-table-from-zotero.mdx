import { Check, Step } from "@/components/learn/lesson-mdx";
import { UiLabel } from "@/components/ui-label";
import { m } from "@/paraglide/messages.js";

<Step n={1}>

In the <UiLabel name={m.library_view_name()} /> tab, select the `Reproducibility` collection in the list on the left. Select <UiLabel name={m.annot_view_save_as_query_file()} />, the page icon with a plus at the top right of the list. Type `Thesis` as the folder, then select <UiLabel name={m.annot_view_save_action()} />.

<Check>
  The dialog reads{" "}
  <UiLabel
    name={m.annot_view_save_file({ file: "Thesis/Reproducibility.zotquery" })}
  />
  . After selecting <UiLabel name={m.annot_view_save_action()} />, the file
  opens in a new tab and lists the papers from your collection.
</Check>

</Step>

<Step n={2}>

Select <UiLabel name={m.query_view_views()} /> at the top left of the tab, then select <UiLabel name={m.query_view_configure_view()} />, the arrow at the end of the row. Set <UiLabel name={m.query_view_layout()} /> to <UiLabel name={m.query_view_layout_table()} />.

<Check>
  The papers appear as rows with four columns: **Item Type**, **Title**,
  **Modified**, and <UiLabel name={m.query_view_table_literature_note()} />.
</Check>

</Step>

<Step n={3}>

Select <UiLabel name={m.query_view_properties()} />. Clear the check boxes for **Item Type** and **Modified**. Type `year` and select **Year**. Type `publication` and select **Publication**, which holds the journal. Type `creators` and select <UiLabel name={m.query_view_properties_add({ name: "creators" })} />.

<Check>
  The visible columns are **Title**,{" "}
  <UiLabel name={m.query_view_table_literature_note()} />, **Year**,
  **Publication**, and **Creators** (the authors).
</Check>

</Step>

<Step n={4}>

Select <UiLabel name={m.query_view_properties()} /> and type `rating`. Select the second "Add" entry, <UiLabel name={m.query_view_properties_add_vault({ name: "rating" })} />. In each paper's cell under the new column, type a rating, such as `5`, and press Enter.

<Check>
  A new column named **rating (note)** shows your ratings. Entering the first
  rating for a paper without a note displays the notice{" "}
  <UiLabel name={m.notice_created_note()} />
</Check>

</Step>

<Step n={5}>

Select <UiLabel name={m.query_view_sort()} />, then select <UiLabel name={m.query_view_sort_add()} />. Set the first box to <UiLabel name={m.query_view_property_date()} /> and the second box to <UiLabel name={m.query_view_sort_old_new()} />.

<Check>
  The papers reorder from the oldest publication date to the newest.
</Check>

</Step>

<Step n={6}>

Select the result count in the toolbar, such as <UiLabel name={m.query_view_results({ count: 4 })} />. Select <UiLabel name={m.query_view_copy()} />. Open your Chapter 3 draft and paste.

<Check>
  The table appears in the draft with all columns in the exact order you
  arranged.
</Check>

</Step>
