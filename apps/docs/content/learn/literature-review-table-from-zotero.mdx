import { Check, Step } from "@/components/learn/lesson-mdx";
import { UiLabel } from "@/components/ui-label";
import { m } from "@/paraglide/messages.js";

<Step n={1}>

In <UiLabel name={m.library_view_name()} />, select the `Reproducibility` collection, then <UiLabel name={m.annot_view_save_as_query_file()} /> (page-plus icon, top right of the list). Type `Thesis` as the folder and select <UiLabel name={m.annot_view_save_action()} />.

<Check>
  The dialog reads{" "}
  <UiLabel
    name={m.annot_view_save_file({ file: "Thesis/Reproducibility.zotquery" })}
  />
  , and the file opens in a new tab listing your collection’s papers.
</Check>

</Step>

<Step n={2}>

Select <UiLabel name={m.query_view_views()} /> at the top left, then the arrow at the end of the row (<UiLabel name={m.query_view_configure_view()} />). Set <UiLabel name={m.query_view_layout()} /> to <UiLabel name={m.query_view_layout_table()} />.

<Check>
  The papers appear as rows with columns **Item Type**, **Title**, **Modified**,
  and <UiLabel name={m.query_view_table_literature_note()} />.
</Check>

</Step>

<Step n={3}>

Select <UiLabel name={m.query_view_properties()} /> and clear the check boxes for **Item Type** and **Modified**. Type each name and select its entry: `year` for **Year**, `publication` for **Publication**, and `creators` for <UiLabel name={m.query_view_properties_add({ name: "creators" })} />.

<Check>
  The visible columns are **Title**,{" "}
  <UiLabel name={m.query_view_table_literature_note()} />, **Year**,
  **Publication**, and **Creators**.
</Check>

</Step>

<Step n={4}>

Select <UiLabel name={m.query_view_properties()} />, type `rating`, and select <UiLabel name={m.query_view_properties_add_vault({ name: "rating" })} />. Under the new column, type a rating such as `5` in each cell and press Enter.

<Check>
  A new column named **rating (note)** shows your ratings, and entering a rating
  displays <UiLabel name={m.notice_created_note()} />
</Check>

</Step>

<Step n={5}>

Select <UiLabel name={m.query_view_sort()} /> > <UiLabel name={m.query_view_sort_add()} />, set the first box to <UiLabel name={m.query_view_property_date()} /> and the second to <UiLabel name={m.query_view_sort_old_new()} />.

<Check>The table displays the papers sorted by date from old to new.</Check>

</Step>

<Step n={6}>

Select the result count in the toolbar, such as <UiLabel name={m.query_view_results({ count: 4 })} />, then <UiLabel name={m.query_view_copy()} />. Paste the table into your Chapter 3 draft.

<Check>The table appears in the draft with all its columns.</Check>

</Step>
